# Problem 6: 3Sum (LeetCode #15)

**Difficulty:** Medium · **Pattern:** Sorting + two pointers

## Problem Statement

Return all unique triplets `[a, b, c]` in `nums` with `a + b + c = 0`.

## Example

`[-1, 0, 1, 2, -1, -4]` → `[[-1, -1, 2], [-1, 0, 1]]`.

## Constraints

`3 <= nums.length <= 3000`; values are in `[-10^5, 10^5]`.

## Observation

Fixing one number turns the problem into Two Sum. Sorting makes duplicate removal and pointer movement deterministic.

## Learning diagram

```text
sort -> fix first -> left/right pair search -> skip duplicates
```

## Algorithm for the optimal approach

Sort, fix one value, and use two pointers on the remaining suffix.

## Pattern to remember

> Fixed value plus sorted pair target -> two pointers.

## Solution 1: Brute Force

### Observation

Enumerate every triplet and deduplicate it. Time: `O(n^3)`. Space: `O(number of triplets)`.

### Algorithm

1. Try every triple of different indices `i < j < k`.
2. Check whether their values sum to zero.
3. Sort each valid triplet before storing it in a set.
4. Convert the unique set into the answer.

### C++ code

```cpp
class Solution {
   public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        set<vector<int>> uniqueTriplets;

        for (int first = 0; first < static_cast<int>(nums.size()); ++first) {
            for (int second = first + 1; second < static_cast<int>(nums.size());
                 ++second) {
                for (int third = second + 1;
                     third < static_cast<int>(nums.size()); ++third) {
                    if (nums[first] + nums[second] + nums[third] == 0) {
                        vector<int> triplet{nums[first], nums[second],
                                            nums[third]};
                        sort(triplet.begin(), triplet.end());
                        uniqueTriplets.insert(triplet);
                    }
                }
            }
        }

        return {uniqueTriplets.begin(), uniqueTriplets.end()};
    }
};
```

### Complexity

- Time: `O(n^3 log n)` including ordered-set insertion
- Space: output-dependent

## How we derive the optimal solution

```text
Try all triples
      |
      v
Fix one value; the other two must solve Two Sum
      |
      v
Use a hash set for each fixed value: O(n^2) time, O(n) space
      |
      v
Sort the array so pair sums move predictably
      |
      v
Two pointers + duplicate skipping
O(n^2) time, O(1) extra search space
```

## Solution 2: Optimized (Hash Set Two Sum)

For every first index, find two-sum complements in the suffix; store sorted triplets in a set. Time: `O(n^2)`. Space: `O(n)` plus output.

### C++

```cpp
vector<vector<int>> threeSumHash(vector<int>& a) {
    set<vector<int>> out;
    for (int i = 0; i < (int)a.size(); ++i) {
        unordered_set<int> seen;
        for (int j = i + 1; j < (int)a.size(); ++j) {
            int need = -a[i] - a[j];
            if (seen.count(need)) {
                vector<int> t{a[i], a[j], need};
                sort(t.begin(), t.end());
                out.insert(t);
            }
            seen.insert(a[j]);
        }
    }
    return {out.begin(), out.end()};
}
```

### Python

```python
def three_sum_hash(nums: list[int]) -> list[list[int]]:
    out: set[tuple[int, int, int]] = set()
    for i, first in enumerate(nums):
        seen: set[int] = set()
        for value in nums[i + 1 :]:
            if -first - value in seen:
                out.add(tuple(sorted((first, value, -first - value))))
            seen.add(value)
    return [list(item) for item in out]
```

### Go

```go
func threeSumHash(a []int) [][]int {
	out := map[[3]int]bool{}
	for i, first := range a {
		seen := map[int]bool{}
		for _, value := range a[i+1:] {
			need := -first - value
			if seen[need] {
				t := []int{first, value, need}
				sort.Ints(t)
				out[[3]int{t[0], t[1], t[2]}] = true
			}
			seen[value] = true
		}
	}
	answer := [][]int{}
	for t := range out {
		answer = append(answer, []int{t[0], t[1], t[2]})
	}
	return answer
}
```

## Approach 3 — Competitive Programming (Sorted Two Pointers)

Sort, fix the first index, scan the rest from both ends, and skip duplicate values. Time: `O(n^2)`. Extra space: `O(1)` excluding sort/output.

### C++

```cpp
vector<vector<int>> threeSum(vector<int>& a) {
    sort(a.begin(), a.end());
    vector<vector<int>> out;
    for (int i = 0; i < (int)a.size(); ++i) {
        if (i && a[i] == a[i - 1]) continue;
        for (int l = i + 1, r = a.size() - 1; l < r;) {
            int sum = a[i] + a[l] + a[r];
            if (sum < 0)
                ++l;
            else if (sum > 0)
                --r;
            else {
                out.push_back({a[i], a[l++], a[r--]});
                while (l < r && a[l] == a[l - 1]) ++l;
                while (l < r && a[r] == a[r + 1]) --r;
            }
        }
    }
    return out;
}
```

### Python

```python
def three_sum(nums: list[int]) -> list[list[int]]:
    nums.sort()
    answer = []
    for i, first in enumerate(nums):
        if i and first == nums[i - 1]:
            continue
        left, right = i + 1, len(nums) - 1
        while left < right:
            total = first + nums[left] + nums[right]
            if total < 0:
                left += 1
            elif total > 0:
                right -= 1
            else:
                answer.append([first, nums[left], nums[right]])
                left += 1
                right -= 1
                while left < right and nums[left] == nums[left - 1]:
                    left += 1
                while left < right and nums[right] == nums[right + 1]:
                    right -= 1
    return answer
```

### Go

```go
func threeSum(a []int) [][]int {
	sort.Ints(a)
	out := [][]int{}
	for i, first := range a {
		if i > 0 && first == a[i-1] {
			continue
		}
		for l, r := i+1, len(a)-1; l < r; {
			sum := first + a[l] + a[r]
			if sum < 0 {
				l++
			} else if sum > 0 {
				r--
			} else {
				out = append(out, []int{first, a[l], a[r]})
				l++
				r--
				for l < r && a[l] == a[l-1] {
					l++
				}
				for l < r && a[r] == a[r+1] {
					r--
				}
			}
		}
	}
	return out
}
```
